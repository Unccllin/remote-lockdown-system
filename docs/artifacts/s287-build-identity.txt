--- [2] why not just put it in AGENT_VERSION ---
    function compareVersions(a, b) {
      const partsA = a.split('.').map(Number);
      const partsB = b.split('.').map(Number);
      const len = Math.max(partsA.length, partsB.length);
      for (let i = 0; i < len; i++) {
        const va = partsA[i] || 0;
        const vb = partsB[i] || 0;
        if (va < vb) return -1;
        if (va > vb) return 1;
      }
      return 0;
    }
    -> Number() on a suffix yields NaN, so every comparison would silently read as equal.
